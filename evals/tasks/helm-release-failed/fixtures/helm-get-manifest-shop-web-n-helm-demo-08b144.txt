  name: shop-web
  labels:

  replicas: 1
  selector:
    matchLabels:
      app.kubernetes.io/name: shop-web
      app.kubernetes.io/instance: shop-web

    metadata:
      labels:
        app.kubernetes.io/name: shop-web
        app.kubernetes.io/instance: shop-web

          image: "nginx:1.99.99-nope"

            - name: http

          readinessProbe:

              path: /
              port: 80
          resources:
            requests:
              cpu: 20m
              memory: 32Mi
